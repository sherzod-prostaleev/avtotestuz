- [ ] **Step 3: Implement `talaffuz.py`**

```python
"""What the voice says versus what the screen shows.

A caption is written once, in display form. `[display](spoken words)` groups say how
digits and the brand are pronounced; every other word is spoken as shown. Muxlisa reads
the ASCII apostrophe best (blind test 2026-10-01, output/tts-sinov), so all o‘/g‘
apostrophes become "'", and the lexicon respells English names the way Uzbeks say them.
"""
import re

GROUP = re.compile(r'\[([^\]]+)\]\(([^)]+)\)')
PUNCT = '«»—–.,!?:;"'
APOSTROPHES = 'ʻ‘’`'
LEXICON = {'iPhone': 'ayfon', 'Android': 'android'}
SPEAKABLE = re.compile(r"^[A-Za-z' ,.!?:;«»—–-]+$")


def tokens(text):
    """[(display_token, [spoken words])] in reading order."""
    out = []
    pos = 0
    for match in GROUP.finditer(text):
        for word in text[pos:match.start()].split():
            out.append((word, [word]))
        display, spoken = match.group(1), match.group(2)
        pos = match.end()
        # Punctuation glued to the group («[Driver Go](Drayver Go).») belongs to it.
        while pos < len(text) and text[pos] in PUNCT:
            display += text[pos]
            pos += 1
        out.append((display, spoken.split()))
    for word in text[pos:].split():
        out.append((word, [word]))
    return out


def spoken_word(word):
    for mark in APOSTROPHES:
        word = word.replace(mark, "'")
    core = word.strip(PUNCT)
    if core in LEXICON:
        word = word.replace(core, LEXICON[core])
    return word


def speech_text(text):
    parts = []
    for display, spoken in tokens(text):
        if len(spoken) == 1 and spoken[0] == display:
            parts.append(spoken_word(display))
        else:
            tail = display[len(display.rstrip(PUNCT)):]
            parts.append(' '.join(spoken_word(w) for w in spoken) + tail)
    return ' '.join(parts)


def check_speakable(speech):
    if not SPEAKABLE.match(speech):
        bad = sorted({ch for ch in speech if not SPEAKABLE.match(ch)})
        raise ValueError(f'unspeakable characters {bad} in {speech!r}')
    for name in list(LEXICON) + ['Driver']:
        if re.search(rf'\b{re.escape(name)}\b', speech):
            raise ValueError(f'{name!r} left unconverted in {speech!r}')
```

```python
import sys
import types
import unittest
from pathlib import Path

import numpy as np

from align import speech_bounds, time_tokens

# asr_check imports faster_whisper at module level; fold() never touches it, and the system
# python that runs these tests does not have it. A stub keeps the real fold() under test.
sys.modules.setdefault('faster_whisper', types.SimpleNamespace(WhisperModel=None))
sys.path.insert(0, str(Path(__file__).resolve().parents[2] / 'reklama-02'))
from asr_check import fold as uz_fold  # noqa: E402


def simple_fold(text):
    return ''.join(ch for ch in text.lower() if ch.isalpha()).split() or []


def heard(*items):
    return [{'word': w, 'start': s, 'end': e} for w, s, e in items]


class TimeTokensTest(unittest.TestCase):
    def test_exact_match_takes_whisper_times(self):
        toks = [('Salom', ['Salom']), ('dunyo!', ['dunyo!'])]
        out, coverage = time_tokens(toks, heard(('Salom', 0.1, 0.5), ('dunyo', 0.6, 1.0)), simple_fold, 0.1, 1.0)
        self.assertEqual([(t['display'], t['start'], t['end']) for t in out],
                         [('Salom', 0.1, 0.5), ('dunyo!', 0.6, 1.0)])
        self.assertEqual(coverage, 1.0)

    def test_missing_word_is_interpolated_by_length(self):
        toks = [('aa', ['aa']), ('bbbb', ['bbbb']), ('cc', ['cc'])]
        out, coverage = time_tokens(toks, heard(('aa', 0.0, 1.0), ('cc', 2.0, 3.0)), simple_fold, 0.0, 3.0)
        self.assertEqual((out[1]['start'], out[1]['end']), (1.0, 2.0))
        self.assertAlmostEqual(coverage, 2 / 3)

    def test_two_heard_pieces_match_one_script_word(self):
        toks = [('imtihonga', ['imtihonga'])]
        out, _ = time_tokens(toks, heard(('imtihon', 0.2, 0.6), ('ga', 0.6, 0.8)), simple_fold, 0.2, 0.8)
        self.assertEqual((out[0]['start'], out[0]['end']), (0.2, 0.8))

    def test_one_heard_word_splits_over_two_script_words(self):
        toks = [('ab', ['ab']), ('cdef', ['cdef'])]
        out, coverage = time_tokens(toks, heard(('abcdef', 0.0, 0.6)), simple_fold, 0.0, 0.6)
        self.assertAlmostEqual(out[0]['end'], 0.2)
        self.assertEqual(coverage, 1.0)

    def test_group_token_spans_its_spoken_words(self):
        toks = [('Driver Go.', ['Drayver', 'Go'])]
        out, _ = time_tokens(toks, heard(('Drayver', 0.0, 0.4), ('Go', 0.45, 0.7)), simple_fold, 0.0, 0.7)
        self.assertEqual((out[0]['start'], out[0]['end']), (0.0, 0.7))

    def test_dash_borrows_the_gap(self):
        toks = [('xato', ['xato']), ('—', ['—']), ('va', ['va'])]
        out, coverage = time_tokens(toks, heard(('xato', 0.0, 0.4), ('va', 0.7, 0.9)), simple_fold, 0.0, 0.9)
        self.assertEqual((out[1]['start'], out[1]['end']), (0.4, 0.7))
        self.assertEqual(coverage, 1.0)

    def test_whisper_spelling_still_matches_with_uzbek_fold(self):
        toks = [('Xatolaringiz', ['Xatolaringiz']), ('yo‘qolmaydi:', ["yo'qolmaydi:"])]
        out, coverage = time_tokens(toks, heard(('Xatalaringiz', 0.0, 0.7), ('yaqalmaydi', 0.75, 1.3)),
                                    uz_fold, 0.0, 1.3)
        self.assertEqual(coverage, 1.0)
        self.assertEqual(out[1]['start'], 0.75)
```

```python
"""Karaoke word times for a TTS that reports none (Muxlisa returns only a .wav).

Whisper large-v3 hears the line with word timestamps; its spelling of Uzbek is loose,
so both sides are folded (reklama-02/asr_check.fold) and matched by a small dynamic
program. One script word may match one or two heard pieces, and two script words one
heard piece. Words Whisper missed get times interpolated by length between their matched
neighbours. Coverage reports how much was really heard; voice.py refuses a line below 0.7.
"""
from difflib import SequenceMatcher

import numpy as np

THRESHOLD = 0.5          # one script word ↔ one heard word
MULTI_THRESHOLD = 0.75   # two-to-one merges must be near-certain…
MULTI_PENALTY = 0.5      # …and cost something, or «aa bbbb» would swallow a lone «aa»
GAP = -0.2
NEG = float('-inf')


def speech_bounds(samples, sr, floor_db=-40.0):
    hop = max(1, sr // 100)
    frames = len(samples) // hop
    rms = np.sqrt(np.mean(samples[:frames * hop].reshape(frames, hop).astype(np.float64) ** 2, axis=1) + 1e-12)
    loud = np.nonzero(20 * np.log10(rms / rms.max()) > floor_db)[0]
    return round(loud[0] * hop / sr, 3), round((loud[-1] + 1) * hop / sr, 3)


def _key(fold, text):
    return ''.join(fold(text))


def _similarity(a, b):
    return SequenceMatcher(None, a, b).ratio() if a and b else 0.0


def _match(script, heard):
    """Per script word: (start, end) from Whisper, or None."""
    n, m = len(script), len(heard)
    best = [[NEG] * (m + 1) for _ in range(n + 1)]
    back = [[None] * (m + 1) for _ in range(n + 1)]
    best[0][0] = 0.0
    for i in range(n + 1):
        for j in range(m + 1):
            here = best[i][j]
            if here == NEG:
                continue
            moves = []
            if i < n:
                moves.append((i + 1, j, GAP, None))
            if j < m:
                moves.append((i, j + 1, GAP, None))
            for ds, da in ((1, 1), (1, 2), (2, 1)):
                if i + ds <= n and j + da <= m:
                    sim = _similarity(''.join(script[i:i + ds]), ''.join(h['key'] for h in heard[j:j + da]))
                    size = max(ds, da)
                    if sim >= (THRESHOLD if size == 1 else MULTI_THRESHOLD):
                        moves.append((i + ds, j + da, sim * size - MULTI_PENALTY * (size - 1), (ds, da)))
            for ni, nj, gain, step in moves:
                if here + gain > best[ni][nj]:
                    best[ni][nj] = here + gain
                    back[ni][nj] = (i, j, step)
    times = [None] * n
    i, j = n, m
    while (i, j) != (0, 0):
        pi, pj, step = back[i][j]
        if step:
            ds, da = step
            start, end = heard[pj]['start'], heard[pj + da - 1]['end']
            if ds == 1:
                times[pi] = (start, end)
            else:
                share = max(len(script[pi]), 1) / (max(len(script[pi]), 1) + max(len(script[pi + 1]), 1))
                middle = start + (end - start) * share
                times[pi], times[pi + 1] = (start, middle), (middle, end)
        i, j = pi, pj
    return times


def _fill(script, times, start, end):
    filled = list(times)
    k = 0
    while k < len(filled):
        if filled[k] is not None:
            k += 1
            continue
        stop = k
        while stop < len(filled) and filled[stop] is None:
            stop += 1
        left = filled[k - 1][1] if k else start
        right = filled[stop][0] if stop < len(filled) else end
        right = max(right, left)
        weights = [max(len(script[x]), 1) for x in range(k, stop)]
        cursor = left
        for x, weight in zip(range(k, stop), weights):
            step = (right - left) * weight / sum(weights)
            filled[x] = (cursor, cursor + step)
            cursor += step
        k = stop
    return filled


def time_tokens(toks, heard, fold, start, end):
    words = []                      # (token index, folded key) for every alignable spoken word
    for index, (_, spoken) in enumerate(toks):
        for word in spoken:
            key = _key(fold, word)
            if key:
                words.append((index, key))
    pieces = [{'key': _key(fold, h['word']), 'start': h['start'], 'end': h['end']} for h in heard]
    pieces = [p for p in pieces if p['key']]
    script = [key for _, key in words]
    matched = _match(script, pieces)
    coverage = sum(t is not None for t in matched) / len(script) if script else 1.0
    filled = _fill(script, matched, start, end)
    spans = {}
    for (index, _), (s, e) in zip(words, filled):
        first, last = spans.get(index, (s, e))
        spans[index] = (min(first, s), max(last, e))
    out = []
    for index, (display, _) in enumerate(toks):
        s, e = spans.get(index, (None, None))
        out.append({'display': display, 'start': s, 'end': e})
    # Tokens with nothing to say (a dash) borrow the gap around them.
    for k, item in enumerate(out):
        if item['start'] is None:
            before = next((r['end'] for r in reversed(out[:k]) if r['end'] is not None), start)
            after = next((r['start'] for r in out[k + 1:] if r['start'] is not None), before)
            item['start'], item['end'] = before, after
    for item in out:
        item['start'], item['end'] = round(item['start'], 3), round(item['end'], 3)
    return out, coverage
```

```python
"""Voice-over for reklama-03: Muxlisa AI «Asomiddin» (speaker 1), one request per line.

Muxlisa returns a bare .wav, so word times come from Whisper large-v3 aligned to the
script (align.py). Lines already on disk are reused (Muxlisa bills per character); pass

import argparse
import importlib.util
import json
import subprocess
import sys
from pathlib import Path

sys.path.insert(0, str(OUT.parent / 'reklama-02'))
```

```python
"""Timeline for reklama-03, derived from the voice-over (same schema as reklama-02).

Every line's first word is snapped onto an eighth-note grid of a 100 BPM score, so
scene changes, caption pages and musical hits share one clock. Writes assets/timeline.json.
"""
import json
import math
from pathlib import Path

OUT = Path(__file__).resolve().parent
ASSETS = OUT / 'assets'
BPM = 100
BEAT = 60 / BPM
GRID = BEAT / 2
BAR = 4 * BEAT
GAP_SAME_SCENE = 0.28
GAP_NEW_SCENE = 0.5
LEAD = 0.14            # visuals start moving just before the first word of a scene
OUTRO = 1.5


def snap_up(t, grid):
    return math.ceil(t / grid - 1e-9) * grid


def build():
    voice = json.loads((ASSETS / 'voice' / 'voice.json').read_text())
    lines = []
    cursor = 0.0
    previous_scene = None
    for line in voice['lines']:
        offset = line['speech_start']
        if not lines:
            speech_at = offset                      # the hook speaks from the first frame
        else:
            gap = GAP_SAME_SCENE if line['scene'] == previous_scene else GAP_NEW_SCENE
            speech_at = snap_up(cursor + gap, GRID)
        start = speech_at - offset
        entry = {
            'id': line['id'], 'scene': line['scene'], 'file': line['file'], 'caption': line['caption'],
            'audio_at': round(start, 4),
            'speech_start': round(speech_at, 4),
            'speech_end': round(start + line['speech_end'], 4),
            'tokens': [{'display': tok['display'], 'start': round(start + tok['start'], 4),
                        'end': round(start + tok['end'], 4)} for tok in line['tokens']],
        }
        assert entry['speech_start'] >= (lines[-1]['speech_end'] if lines else 0) - 1e-6, entry['id']
        lines.append(entry)
        cursor = entry['speech_end']
        previous_scene = line['scene']
    scenes = {}
    for entry in lines:
        if entry['scene'] not in scenes:
            scenes[entry['scene']] = round(entry['speech_start'] - LEAD, 4) if scenes else 0.0
    duration = snap_up(lines[-1]['speech_end'] + OUTRO, BEAT)
    return {'bpm': BPM, 'beat': BEAT, 'bar': BAR, 'grid': GRID, 'duration': round(duration, 4),
            'voice': voice['voice'], 'rate': voice['rate'], 'scenes': scenes, 'lines': lines}


def main():
    plan = build()
    (ASSETS / 'timeline.json').write_text(json.dumps(plan, ensure_ascii=False, indent=1))
    for entry in plan['lines']:
        print(f"{entry['id']:10s} {entry['scene']:12s} {entry['speech_start']:6.2f} → {entry['speech_end']:6.2f}")
    print(f"duration {plan['duration']:.2f}s = {plan['duration'] / plan['beat']:.0f} beats at {plan['bpm']} BPM")


if __name__ == '__main__':
    main()
```
